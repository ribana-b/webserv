#!/bin/bash
# Script para instalar el driver oficial de Realtek r8125/r8126
# Ejecutar con: sudo bash install_realtek_driver.sh

set -e

echo "=== INSTALACIÓN DEL DRIVER OFICIAL REALTEK 8126 ==="
echo ""

# Verificar que se ejecuta como root
if [ "$EUID" -ne 0 ]; then
    echo "❌ ERROR: Este script debe ejecutarse como root"
    echo "   Usa: sudo bash install_realtek_driver.sh"
    exit 1
fi

# Verificar dependencias
echo "[1/8] Verificando dependencias..."
if ! dpkg -l | grep -q "build-essential\|linux-headers\|dkms"; then
    echo "Instalando dependencias..."
    apt update
    apt install -y build-essential dkms linux-headers-$(uname -r)
fi
echo "✓ Dependencias verificadas"
echo ""

# Crear directorio temporal
WORK_DIR="/tmp/realtek_driver_install"
mkdir -p "$WORK_DIR"
cd "$WORK_DIR"

echo "[2/8] Descargando driver oficial de Realtek 8126..."
# URL del driver oficial (versión más reciente)
DRIVER_URL="https://github.com/awesometic/realtek-r8125-dkms/archive/refs/heads/main.zip"

if command -v wget &> /dev/null; then
    wget -O realtek-driver.zip "$DRIVER_URL"
elif command -v curl &> /dev/null; then
    curl -L -o realtek-driver.zip "$DRIVER_URL"
else
    echo "❌ ERROR: wget o curl no encontrados"
    exit 1
fi
echo "✓ Driver descargado"
echo ""

echo "[3/8] Extrayendo archivos..."
apt install -y unzip
unzip -q realtek-driver.zip
cd realtek-r8125-dkms-main
echo "✓ Archivos extraídos"
echo ""

echo "[4/8] Respaldando configuración actual..."
BACKUP_DIR="/root/network_backup_$(date +%Y%m%d_%H%M%S)"
mkdir -p "$BACKUP_DIR"
lsmod | grep r8169 > "$BACKUP_DIR/modules_before.txt" 2>/dev/null || true
modinfo r8169 > "$BACKUP_DIR/r8169_info.txt" 2>/dev/null || true
ethtool enp8s0 > "$BACKUP_DIR/ethtool_before.txt" 2>/dev/null || true
echo "✓ Backup creado en $BACKUP_DIR"
echo ""

echo "[5/8] Instalando driver con DKMS..."
# Copiar archivos DKMS
DKMS_DIR="/usr/src/r8125-9.013.02"
cp -r . "$DKMS_DIR"

# Registrar con DKMS
dkms add -m r8125 -v 9.013.02
dkms build -m r8125 -v 9.013.02
dkms install -m r8125 -v 9.013.02
echo "✓ Driver instalado con DKMS"
echo ""

echo "[6/8] Agregando a blacklist el driver antiguo r8169..."
cat > /etc/modprobe.d/realtek-blacklist.conf <<EOF
# Blacklist del driver genérico r8169 para usar r8125
blacklist r8169
EOF
echo "✓ r8169 agregado a blacklist"
echo ""

echo "[7/8] Actualizando initramfs..."
update-initramfs -u
echo "✓ Initramfs actualizado"
echo ""

echo "[8/8] Creando script de rollback..."
cat > /root/rollback_realtek.sh <<'ROLLBACK'
#!/bin/bash
# Script de rollback por si algo sale mal
echo "=== ROLLBACK DEL DRIVER REALTEK ==="
dkms remove -m r8125 -v 9.013.02 --all
rm -f /etc/modprobe.d/realtek-blacklist.conf
update-initramfs -u
echo "✓ Rollback completado. Reinicia el sistema."
ROLLBACK
chmod +x /root/rollback_realtek.sh
echo "✓ Script de rollback creado en /root/rollback_realtek.sh"
echo ""

echo "=========================================="
echo "✓ INSTALACIÓN COMPLETADA"
echo ""
echo "IMPORTANTE: Debes REINICIAR el sistema para aplicar los cambios"
echo ""
echo "Después del reinicio:"
echo "  1. Verifica el driver: lsmod | grep r8125"
echo "  2. Verifica velocidad: ethtool enp8s0 | grep Speed"
echo "  3. Haz un speedtest: speedtest-cli --secure"
echo ""
echo "Si algo sale mal, ejecuta:"
echo "  sudo bash /root/rollback_realtek.sh"
echo ""
echo "Backup de configuración anterior en: $BACKUP_DIR"
echo "=========================================="
echo ""
read -p "¿Deseas reiniciar ahora? (s/n) " -n 1 -r
echo
if [[ $REPLY =~ ^[SsYy]$ ]]; then
    echo "Reiniciando en 5 segundos..."
    sleep 5
    reboot
else
    echo "Recuerda reiniciar manualmente con: sudo reboot"
fi
